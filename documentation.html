Verification Codes Telegram

Сервис для отправки кодов подтверждения в Telegram через MQTT. Клиент (браузер или Python-скрипт) публикует зашифрованное сообщение в публичный брокер, серверное приложение принимает его и отправляет код нужному пользователю через бота.

Клиенту не нужно запускать сервер, хранить токен бота или знать его username. Достаточно действующего сертификата сервиса и доступа к MQTT-брокеру.

Ключевая особенность: все сообщения шифруются AES-256-GCM с общим ключом. В MQTT передаётся только шифртекст в base64 — посторонний, подключившись к брокеру, не увидит ни сертификат, ни код, ни username получателя.

<script>
(async () => {
  const BROKER_URL     = 'wss://broker.hivemq.com:8884/mqtt';
  const SHARED_KEY_B64 = 'IEhtOQMjzZWW7+sbqQlAX2OpMq1P7MXFuBIYGiGKD2M=';

  const certEl    = document.getElementById('cert');
  const serviceEl = document.getElementById('service');
  const codeEl    = document.getElementById('code');
  const userEl    = document.getElementById('username');
  const sendBtn   = document.getElementById('sendBtn');
  const msgEl     = document.getElementById('msg');

  function setMsg(text, type) {
    msgEl.textContent = text || '';
    msgEl.className = 'msg' + (type ? ' ' + type : '');
  }

  const saved = localStorage.getItem('service_certificate');
  if (saved) certEl.value = saved;
  certEl.addEventListener('change', () => {
    localStorage.setItem('service_certificate', certEl.value.trim());
  });

  if (!window.crypto || !window.crypto.subtle) {
    setMsg('Нужен https:// или http://localhost', 'error');
    return;
  }

  const keyBytes  = Uint8Array.from(atob(SHARED_KEY_B64), c => c.charCodeAt(0));
  const cryptoKey = await crypto.subtle.importKey(
    'raw', keyBytes, { name: 'AES-GCM' }, false, ['encrypt', 'decrypt']
  );
  const digest = await crypto.subtle.digest('SHA-256', keyBytes);
  const PREFIX = Array.from(new Uint8Array(digest))
    .map(b => b.toString(16).padStart(2, '0')).join('').slice(0, 16);

  const SESSION = crypto.randomUUID
    ? crypto.randomUUID()
    : 'sess-' + Math.random().toString(16).slice(2) + Date.now().toString(16);

  const T_SEND     = `${PREFIX}/${SESSION}/send`;
  const T_SEND_ACK = `${PREFIX}/${SESSION}/send-ack`;

  async function encrypt(obj) {
    const iv = crypto.getRandomValues(new Uint8Array(12));
    const ct = await crypto.subtle.encrypt(
      { name: 'AES-GCM', iv }, cryptoKey,
      new TextEncoder().encode(JSON.stringify(obj))
    );
    return {
      iv: btoa(String.fromCharCode(...iv)),
      data: btoa(String.fromCharCode(...new Uint8Array(ct))),
    };
  }

  async function decrypt(envelope) {
    const iv = Uint8Array.from(atob(envelope.iv), c => c.charCodeAt(0));
    const data = Uint8Array.from(atob(envelope.data), c => c.charCodeAt(0));
    const pt = await crypto.subtle.decrypt({ name: 'AES-GCM', iv }, cryptoKey, data);
    return JSON.parse(new TextDecoder().decode(pt));
  }

  let mqttReady = false;
  const client = mqtt.connect(BROKER_URL, {
    clientId: ('html-' + SESSION).slice(0, 23),
    clean: true,
    reconnectPeriod: 2000,
    connectTimeout: 10000,
  });

  client.on('connect', () => {
    client.subscribe(T_SEND_ACK, { qos: 1 }, (err) => {
      if (err) { setMsg('Ошибка подписки MQTT', 'error'); return; }
      mqttReady = true;
      setMsg('Подключено к брокеру');
    });
  });
  client.on('reconnect', () => setMsg('Переподключение...'));
  client.on('error', (e) => setMsg('MQTT: ' + e.message, 'error'));

  client.on('message', async (topic, payload) => {
    if (topic !== T_SEND_ACK) return;
    let data;
    try {
      data = await decrypt(JSON.parse(payload.toString()));
    } catch (e) {
      setMsg('Не удалось расшифровать ответ', 'error');
      sendBtn.disabled = false;
      return;
    }
    sendBtn.disabled = false;
    if (data.ok) setMsg('Код отправлен в Telegram', 'ok');
    else         setMsg(data.error || 'Ошибка отправки', 'error');
  });

  sendBtn.addEventListener('click', async () => {
    if (!mqttReady) { setMsg('Нет связи с брокером', 'error'); return; }

    const certificate  = certEl.value.trim();
    const service_name = serviceEl.value.trim();
    const code         = codeEl.value.trim();
    const username     = userEl.value.trim();

    if (!certificate)  { setMsg('Введите сертификат сервиса', 'error'); return; }
    if (certificate.length !== 30) { setMsg('Сертификат должен быть 30 символов', 'error'); return; }
    if (!service_name) { setMsg('Введите название сервиса', 'error'); return; }
    if (!code)         { setMsg('Введите код', 'error'); return; }
    if (!username)     { setMsg('Введите username', 'error'); return; }

    localStorage.setItem('service_certificate', certificate);

    sendBtn.disabled = true;
    setMsg('Отправка...');

    const env = await encrypt({ certificate, service_name, code, username });
    client.publish(T_SEND, JSON.stringify(env), { qos: 1 });

    setTimeout(() => {
      if (sendBtn.disabled) {
        sendBtn.disabled = false;
        setMsg('Сервер не ответил', 'error');
      }
    }, 15000);
  });
})();
</script>

Как запустить: сохраните код выше в файл index.html, в папке с ним выполните python -m http.server 8080 и откройте http://localhost:8080. Либо разместите на любом статичном хостинге по HTTPS.

Что делает эта страница

  1. Вычисляет PREFIX = SHA256(SHARED_KEY)[:16] для топиков.
  2. Генерирует случайный SESSION (UUID) — уникальный для каждого открытия.
  3. Подключается к wss://broker.hivemq.com:8884/mqtt и подписывается на {PREFIX}/{SESSION}/send-ack.
  4. По кнопке «Отправить» шифрует {certificate, service_name, code, username} через AES-256-GCM и публикует в {PREFIX}/{SESSION}/send.
  5. Получает зашифрованный ответ, расшифровывает и показывает «Код отправлен в Telegram» либо текст ошибки.

Сертификат сервиса

Сертификат — строка из ровно 30 символов, выданная владельцем сервиса через админ-панель. Без него сервер отклоняет любой запрос.

Алфавит сертификата

ABCDEFGHIJKLMNOPQRSTUVWXYZ
abcdefghijklmnopqrstuvwxyz
0123456789
!@#$%^&*-_=+

Все символы допустимы в любой позиции, длина — строго 30.

Правила использования

Сервер возвращает ошибку У сервиса невалидный или истёкший код сертификата, если:

  • сертификат не существует;
  • сертификат отключён владельцем;
  • истёк срок действия;
  • исчерпан лимит отправок;
  • сертификат не передан в запросе.

Лимиты

ПараметрЗначение
Срок действиядата, заданная владельцем, либо бессрочно
Лимит отправокчисло, заданное владельцем, либо безлимит
Анти-спамне чаще 1 отправки одному username в 30 секунд

MQTT-топики

Брокер: wss://broker.hivemq.com:8884/mqtt (браузер) или broker.hivemq.com:1883 (Python).

Все топики имеют вид:

<PREFIX>/<SESSION>/<ACTION>
ПолеОписание
PREFIXSHA256(SHARED_KEY)[:16] — 16 шестнадцатеричных символов, одинаковые у всех клиентов с правильным ключом
SESSIONУникальный идентификатор сессии (UUID или случайная строка). Одна сессия = одна пара «запрос — ответ»
ACTIONsend — запрос отправки; send-ack — ответ сервера

Список топиков

ТопикНаправлениеНазначение
{PREFIX}/{SESSION}/sendклиент → серверзапрос на отправку кода
{PREFIX}/{SESSION}/send-ackсервер → клиентрезультат отправки

Вычисление PREFIX

import base64, hashlib

KEY    = base64.b64decode("IEhtOQMjzZWW7+sbqQlAX2OpMq1P7MXFuBIYGiGKD2M=")
PREFIX = hashlib.sha256(KEY).hexdigest()[:16]
# например, "a1b2c3d4e5f60718"

Формат сообщений

Внутри MQTT передаётся JSON-обёртка с зашифрованными данными:

{
  "iv":   "base64(12-байтный случайный IV)",
  "data": "base64(AES-256-GCM-шифртекст)"
}

Расшифрованный data — обычный JSON. Структура зависит от направления.

Запрос отправки (send)

ПолеТипОбязательноОписание
certificatestringда30-символьный сертификат
service_namestringданазвание сервиса для текста в Telegram
codestringдакод подтверждения, до 64 символов
usernamestringда@username или username

Ответ сервера (send-ack)

ПолеТипОписание
okbooltrue — отправлено, false — ошибка
errorstringтекст ошибки (только при ok: false)

Пример расшифрованного запроса

{
  "certificate":  "Ab3!Kx9_Qw*Zt7-Rm2#Lp5+Ns8=Yu4&Hj6@Vd1Eo",
  "service_name": "Сбербанк Онлайн",
  "code":         "482913",
  "username":     "@ivan_petrov"
}

Пример расшифрованного ответа

{"ok": true}
{"ok": false, "error": "У сервиса невалидный или истёкший код сертификата"}

Шифрование

Каждое сообщение шифруется AES-256-GCM с общим ключом (32 байта, base64). Ключ уже находится в HTML-файлах и серверном коде.

ПараметрЗначение
АлгоритмAES-GCM
Длина ключа256 бит (32 байта)
IV12 байт, случайный на каждое сообщение
Кодирование IVbase64
Кодирование шифртекстаbase64
Тег аутентификациивстроен в шифртекст (GCM)

Шаги шифрования

  1. Сериализовать JSON-объект в UTF-8 байты.
  2. Сгенерировать 12 случайных байт (IV).
  3. Зашифровать байты AES-GCM с этим IV.
  4. Собрать {"iv": base64(iv), "data": base64(ct)}.
  5. Опубликовать в нужный топик.

Расшифровка ответа

  1. Распарсить JSON, получить iv и data.
  2. Декодировать оба из base64.
  3. Расшифровать AES-GCM с тем же ключом и IV.
  4. Распарсить получившийся JSON.

JavaScript (браузер)

Минимальный пример без UI: подключение, шифрование, публикация и ожидание ответа.

// <script src="https://unpkg.com/mqtt@5/dist/mqtt.min.js"></script>
(async () => {
  const BROKER_URL     = "wss://broker.hivemq.com:8884/mqtt";
  const SHARED_KEY_B64 = "IEhtOQMjzZWW7+sbqQlAX2OpMq1P7MXFuBIYGiGKD2M=";

  const keyBytes  = Uint8Array.from(atob(SHARED_KEY_B64), c => c.charCodeAt(0));
  const cryptoKey = await crypto.subtle.importKey(
    "raw", keyBytes, { name: "AES-GCM" }, false, ["encrypt", "decrypt"]
  );
  const digest = await crypto.subtle.digest("SHA-256", keyBytes);
  const PREFIX = Array.from(new Uint8Array(digest))
    .map(b => b.toString(16).padStart(2, "0")).join("").slice(0, 16);

  const SESSION = crypto.randomUUID();
  const T_SEND     = `${PREFIX}/${SESSION}/send`;
  const T_SEND_ACK = `${PREFIX}/${SESSION}/send-ack`;

  async function encrypt(obj) {
    const iv = crypto.getRandomValues(new Uint8Array(12));
    const ct = await crypto.subtle.encrypt(
      { name: "AES-GCM", iv }, cryptoKey,
      new TextEncoder().encode(JSON.stringify(obj))
    );
    return {
      iv:   btoa(String.fromCharCode(...iv)),
      data: btoa(String.fromCharCode(...new Uint8Array(ct))),
    };
  }
  async function decrypt(envelope) {
    const iv   = Uint8Array.from(atob(envelope.iv),   c => c.charCodeAt(0));
    const data = Uint8Array.from(atob(envelope.data), c => c.charCodeAt(0));
    const pt = await crypto.subtle.decrypt({ name: "AES-GCM", iv }, cryptoKey, data);
    return JSON.parse(new TextDecoder().decode(pt));
  }

  const client = mqtt.connect(BROKER_URL, {
    clientId: ("html-" + SESSION).slice(0, 23),
  });

  client.on("connect", () => {
    client.subscribe(T_SEND_ACK, { qos: 1 }, async () => {
      const envelope = await encrypt({
        certificate:  "Ab3!Kx9_Qw*Zt7-Rm2#Lp5+Ns8=Yu4&Hj6@Vd1Eo",
        service_name: "Сбербанк Онлайн",
        code:         "482913",
        username:     "@ivan_petrov",
      });
      client.publish(T_SEND, JSON.stringify(envelope), { qos: 1 });
    });
  });

  client.on("message", async (topic, payload) => {
    if (topic !== T_SEND_ACK) return;
    const data = await decrypt(JSON.parse(payload.toString()));
    if (data.ok) console.log("Отправлено");
    else         console.error("Ошибка:", data.error);
  });
})();

Python

Тот же сценарий на Python: paho-mqtt для транспорта, cryptography для AES-256-GCM.

import base64
import hashlib
import json
import os
import time
import uuid

import paho.mqtt.client as mqtt
from cryptography.hazmat.primitives.ciphers.aead import AESGCM

BROKER_HOST    = "broker.hivemq.com"
BROKER_PORT    = 1883
SHARED_KEY_B64 = "IEhtOQMjzZWW7+sbqQlAX2OpMq1P7MXFuBIYGiGKD2M="

KEY    = base64.b64decode(SHARED_KEY_B64)
AES    = AESGCM(KEY)
PREFIX = hashlib.sha256(KEY).hexdigest()[:16]


def encrypt(obj: dict) -> dict:
    iv = os.urandom(12)
    ct = AES.encrypt(iv, json.dumps(obj).encode(), None)
    return {
        "iv":   base64.b64encode(iv).decode(),
        "data": base64.b64encode(ct).decode(),
    }


def decrypt(envelope: dict) -> dict:
    iv = base64.b64decode(envelope["iv"])
    ct = base64.b64decode(envelope["data"])
    return json.loads(AES.decrypt(iv, ct, None).decode())


SESSION    = str(uuid.uuid4())
T_SEND     = f"{PREFIX}/{SESSION}/send"
T_SEND_ACK = f"{PREFIX}/{SESSION}/send-ack"


def on_connect(client, userdata, flags, rc):
    client.subscribe(T_SEND_ACK, qos=1)


def on_message(client, userdata, msg):
    data = decrypt(json.loads(msg.payload.decode()))
    if data.get("ok"):
        print("Отправлено")
    else:
        print("Ошибка:", data.get("error"))
    client.disconnect()


client = mqtt.Client(mqtt.CallbackAPIVersion.VERSION1,
                     client_id=f"cli-{SESSION[:8]}")
client.on_connect = on_connect
client.on_message = on_message
client.connect(BROKER_HOST, BROKER_PORT, 60)
client.loop_start()

time.sleep(1)
envelope = encrypt({
    "certificate":  "Ab3!Kx9_Qw*Zt7-Rm2#Lp5+Ns8=Yu4&Hj6@Vd1Eo",
    "service_name": "Сбербанк Онлайн",
    "code":         "482913",
    "username":     "@ivan_petrov",
})
client.publish(T_SEND, json.dumps(envelope), qos=1)

time.sleep(5)
client.loop_stop()

Частые проблемы

Браузер пишет «Нужен https:// или http://localhost»

WebCrypto (crypto.subtle) не работает на file:// в части браузеров. Поднимите статичный сервер в папке с HTML:

python -m http.server 8080

и откройте http://localhost:8080.

Ответ не приходит

  • PREFIX в клиенте совпадает с тем, на который подписан сервер (одинаковый ключ → одинаковый хеш, должно совпасть автоматически).
  • SESSION в топике запроса и топике подписки одинаковый.
  • Клиент подписан на send-ack до публикации send.
  • Брокер доступен: соединение к wss://broker.hivemq.com:8884/mqtt.

Приходит ошибка о невалидном сертификате

  • Проверьте, что сертификат ровно 30 символов и скопирован без пробелов.
  • Проверьте, что владелец не отключил или не удалил сертификат.
  • Проверьте срок действия и оставшийся лимит отправок у владельца.

Сообщение приходит, но кода в нём нет

Скорее всего, поле code отправлено пустым. Проверьте, что в расшифрованном JSON поле code содержит именно то, что вы ожидаете.

«Forbidden: bot can't initiate conversation with a user»

Получатель никогда не открывал чат с ботом. Попросите его нажать /start у бота и повторите отправку.

«chat not found»

  • Пользователь не нажимал /start в последние часы (поиск chat_id через getUpdates не нашёл его).
  • Username введён с ошибкой.
  • Пользователь скрыл username в настройках приватности.
